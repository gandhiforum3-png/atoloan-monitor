```python
from langchain_anthropic import ChatAnthropic
from langchain_core.messages import SystemMessage, HumanMessage, AIMessage, ToolMessage

# Construction (model id stays "claude-sonnet-4-6" as currently hardcoded
# in diagnoser_base.py / agent_loop.py — do not change the model string).
llm = ChatAnthropic(model="claude-sonnet-4-6", max_tokens=2048)

# --- diagnose_with_claude: forced single-tool structured output ---
# with_structured_output accepts a raw JSON schema dict (result_model.model_json_schema()
# or the per-domain input_schema override) and, with include_raw=True, returns:
#   {"raw": AIMessage, "parsed": <result_model instance or None>, "parsing_error": Exception | None}
structured_llm = llm.with_structured_output(schema_dict, include_raw=True)
out = await structured_llm.ainvoke([
    SystemMessage(content=[{
        "type": "text",
        "text": system_prompt,
        "cache_control": {"type": "ephemeral"},
    }]),
    HumanMessage(content=user_text),
])
raw_msg: AIMessage = out["raw"]
parsed = out["parsed"]          # dict (since schema is a plain dict, not a Pydantic class) -> validate with result_model.model_validate(parsed)
parsing_error = out["parsing_error"]

# Cache token usage on the raw AIMessage:
usage = raw_msg.usage_metadata or {}                       # UsageMetadata TypedDict
input_details = usage.get("input_token_details", {}) or {} # InputTokenDetails TypedDict
cache_read = input_details.get("cache_read", 0)
cache_creation = input_details.get("cache_creation", 0)
input_tokens = usage.get("input_tokens", 0)
output_tokens = usage.get("output_tokens", 0)

# --- run_tool_loop: bind_tools + AIMessage.tool_calls / ToolMessage ---
bound_llm = llm.bind_tools(tools)  # tools: list[dict] in Anthropic tool-schema shape (unchanged)
response: AIMessage = await bound_llm.ainvoke(messages)
# response.tool_calls -> list[ToolCall], ToolCall = {"name": str, "args": dict, "id": str, "type": "tool_call"}
for tc in response.tool_calls:
    name, tool_input, call_id = tc["name"], tc["args"], tc["id"]
    # ... dispatch ...
    messages.append(ToolMessage(content=json.dumps(result), tool_call_id=call_id))
```

Notes:
- `SystemMessage(content=[...])` accepts the SAME list-of-content-block shape
  (`{"type": "text", "text": ..., "cache_control": {...}}`) as the raw
  Anthropic `system` param — this is how `cache_control` is preserved.
- `messages` for `run_tool_loop` becomes a `list[BaseMessage]`
  (`HumanMessage`/`AIMessage`/`ToolMessage`) instead of `list[dict]`.
  `AIMessage.content` may be `""` or a list of content blocks when there
  are tool calls — append the `AIMessage` object directly to `messages`,
  do not reconstruct it.
- `result_model.model_validate(parsed)` — `parsed` from `with_structured_output`
  with a dict schema is itself a `dict`, so validation must still happen in
  `diagnose_with_claude` (same as today's `result_model.model_validate(block.input)`).
  If `parsing_error` is not None, raise `RuntimeError` (equivalent to today's
  "Claude did not return a {tool_name} tool call" branch).
